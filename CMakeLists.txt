cmake_minimum_required(VERSION 3.21)
project(ChessEngineLab VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
option(CHESS_LAB_PROFILE "Preserve native sampling-profiler frame pointers" OFF)
if(CHESS_LAB_PROFILE AND NOT EMSCRIPTEN)
  if(MSVC)
    add_compile_options(/Oy-)
  elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-fno-omit-frame-pointer)
  endif()
endif()

add_library(chess_core STATIC engine/src/build_info.cpp engine/src/chess_core.cpp)
add_library(ChessEngineLab::Core ALIAS chess_core)
target_compile_features(chess_core PUBLIC cxx_std_20)
target_include_directories(chess_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/engine/include")
target_include_directories(chess_core SYSTEM PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/chess-library")
target_compile_definitions(chess_core PRIVATE CHESS_LAB_VERSION="${PROJECT_VERSION}")

# ChessCore's public error contract uses C++ exceptions. Emscripten disables
# catching by default; enable it for the adapter and every linked consumer.
if(EMSCRIPTEN)
  target_compile_options(chess_core PUBLIC "-fexceptions")
  target_link_options(chess_core INTERFACE "-fexceptions")
endif()

if(BUILD_TESTING)
  foreach(test_name IN ITEMS m0_smoke m1_core)
    add_executable(${test_name} tests/${test_name}.cpp)
    target_link_libraries(${test_name} PRIVATE ChessEngineLab::Core)
    if(EMSCRIPTEN)
      find_program(CHESS_LAB_NODE_EXECUTABLE NAMES node nodejs REQUIRED)
      set_target_properties(${test_name} PROPERTIES SUFFIX ".js")
      target_link_options(${test_name} PRIVATE "-sENVIRONMENT=node" "-sEXIT_RUNTIME=1")
      add_test(NAME ${test_name} COMMAND "${CHESS_LAB_NODE_EXECUTABLE}" "$<TARGET_FILE:${test_name}>")
    else()
      add_test(NAME ${test_name} COMMAND ${test_name})
    endif()
    set_tests_properties(${test_name} PROPERTIES TIMEOUT 120)
  endforeach()
endif()

# M2: independently compiled evaluator and host, sharing only the approved C ABI.
add_library(evaluation_support INTERFACE)
target_include_directories(evaluation_support INTERFACE engine/include engine/src/evaluation third_party)
target_compile_features(evaluation_support INTERFACE cxx_std_20)
if(EMSCRIPTEN)
  target_compile_options(evaluation_support INTERFACE -fexceptions)
  target_link_options(evaluation_support INTERFACE -fexceptions)
endif()
add_library(material_provider STATIC engine/src/evaluation/material.cpp)
target_link_libraries(material_provider PRIVATE evaluation_support)
add_library(chess_evaluation STATIC engine/src/evaluation/bridge.cpp engine/src/evaluation/host.cpp)
target_link_libraries(chess_evaluation PUBLIC chess_core PRIVATE evaluation_support material_provider ${CMAKE_DL_LIBS})
target_compile_definitions(chess_evaluation PRIVATE CHESS_LAB_VERSION="${PROJECT_VERSION}" CHESS_LAB_BUILD_TYPE="$<CONFIG>")
if(NOT EMSCRIPTEN)
  add_library(material_evaluator SHARED engine/src/evaluation/material.cpp)
  target_link_libraries(material_evaluator PRIVATE evaluation_support)
  target_compile_definitions(material_evaluator PRIVATE CHESS_LAB_MATERIAL_NATIVE)
  set_target_properties(material_evaluator PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
endif()
if(BUILD_TESTING)
  foreach(test_name IN ITEMS m2_evaluation abi_c abi_cpp)
    if(test_name STREQUAL "abi_c")
      add_executable(${test_name} tests/evaluation/layout.c)
      set_target_properties(${test_name} PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
      target_include_directories(${test_name} PRIVATE engine/include tests/evaluation)
    elseif(test_name STREQUAL "abi_cpp")
      add_executable(${test_name} tests/evaluation/layout.cpp)
      target_link_libraries(${test_name} PRIVATE evaluation_support)
    else()
      add_executable(${test_name} tests/evaluation/m2_evaluation.cpp)
      target_link_libraries(${test_name} PRIVATE chess_evaluation evaluation_support material_provider)
    endif()
    if(EMSCRIPTEN)
      set_target_properties(${test_name} PROPERTIES SUFFIX ".js")
      target_link_options(${test_name} PRIVATE -sENVIRONMENT=node -sEXIT_RUNTIME=1)
      if(test_name STREQUAL "m2_evaluation")
        add_test(NAME ${test_name} COMMAND "${CHESS_LAB_NODE_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/evaluation/run_wasm.cjs" "$<TARGET_FILE:${test_name}>")
      else()
        add_test(NAME ${test_name} COMMAND "${CHESS_LAB_NODE_EXECUTABLE}" "$<TARGET_FILE:${test_name}>")
      endif()
    elseif(test_name STREQUAL "m2_evaluation")
      add_test(NAME ${test_name} COMMAND ${test_name} "$<TARGET_FILE:material_evaluator>" "$<TARGET_FILE:missing_symbol>" "$<TARGET_FILE:incompatible_evaluator>")
    else()
      add_test(NAME ${test_name} COMMAND ${test_name})
    endif()
    set_tests_properties(${test_name} PROPERTIES TIMEOUT 120)
  endforeach()
  if(NOT EMSCRIPTEN)
    add_library(missing_symbol SHARED tests/evaluation/missing_symbol.c)
    add_library(incompatible_evaluator SHARED tests/evaluation/incompatible.cpp)
    target_link_libraries(incompatible_evaluator PRIVATE evaluation_support)
  endif()
  add_executable(m2_benchmark tests/evaluation/benchmark.cpp)
  target_link_libraries(m2_benchmark PRIVATE chess_evaluation evaluation_support material_provider)
endif()

# M3: source-only C++ search API; no third-party types or evaluator ABI changes.
add_library(chess_search STATIC engine/src/search/search.cpp)
add_library(ChessEngineLab::Search ALIAS chess_search)
target_link_libraries(chess_search PUBLIC chess_evaluation)
if(BUILD_TESTING)
  add_executable(m3_search tests/search/m3_search.cpp)
  target_link_libraries(m3_search PRIVATE chess_search evaluation_support material_provider)

    target_link_options(m3_search PRIVATE -sENVIRONMENT=node -sEXIT_RUNTIME=1)
    add_test(NAME m3_search COMMAND "${CHESS_LAB_NODE_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/evaluation/run_wasm.cjs" "$<TARGET_FILE:m3_search>")
  else()
    add_test(NAME m3_search COMMAND m3_search)
  endif()
  set_tests_properties(m3_search PROPERTIES TIMEOUT 300)
endif()
if(NOT EMSCRIPTEN)
  add_executable(m3_benchmark tests/search/benchmark.cpp)
  target_link_libraries(m3_benchmark PRIVATE chess_search)
  target_compile_definitions(m3_benchmark PRIVATE CHESS_LAB_BENCH_BUILD="$<CONFIG>" CHESS_LAB_BENCH_COMPILER="${CMAKE_CXX_COMPILER_ID}-${CMAKE_CXX_COMPILER_VERSION}" CHESS_LAB_BENCH_PLATFORM="${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}")
endif()

# M4 regression suite executes on native and the same pinned WASM/Node harness.
if(BUILD_TESTING)
  add_executable(m4_search tests/search/m4_search.cpp)
  target_link_libraries(m4_search PRIVATE chess_search evaluation_support material_provider)
  if(EMSCRIPTEN)
    set_target_properties(m4_search PROPERTIES SUFFIX ".js")
    target_link_options(m4_search PRIVATE -sENVIRONMENT=node -sEXIT_RUNTIME=1)
    add_test(NAME m4_search COMMAND "${CHESS_LAB_NODE_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/evaluation/run_wasm.cjs" "$<TARGET_FILE:m4_search>")
  else()
    find_package(Threads REQUIRED)
    target_link_libraries(m4_search PRIVATE Threads::Threads)
    add_test(NAME m4_search COMMAND m4_search)
  endif()
  set_tests_properties(m4_search PROPERTIES TIMEOUT 300)
endif()
if(NOT EMSCRIPTEN)
  add_executable(m4_benchmark tests/search/m4_benchmark.cpp)
  target_link_libraries(m4_benchmark PRIVATE chess_search)
  target_compile_definitions(m4_benchmark PRIVATE CHESS_LAB_BENCH_BUILD="$<CONFIG>" CHESS_LAB_BENCH_COMPILER="${CMAKE_CXX_COMPILER_ID}-${CMAKE_CXX_COMPILER_VERSION}" CHESS_LAB_BENCH_PLATFORM="${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}")
endif()
