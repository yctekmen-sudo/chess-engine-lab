cmake_minimum_required(VERSION 3.21)
project(ChessEngineLab VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)

add_library(chess_core STATIC engine/src/build_info.cpp)
add_library(ChessEngineLab::Core ALIAS chess_core)
target_compile_features(chess_core PUBLIC cxx_std_20)
target_include_directories(chess_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/engine/include")
target_compile_definitions(chess_core PRIVATE CHESS_LAB_VERSION="${PROJECT_VERSION}")

if(BUILD_TESTING)
  add_executable(m0_smoke tests/m0_smoke.cpp)
  target_link_libraries(m0_smoke PRIVATE ChessEngineLab::Core)
  if(EMSCRIPTEN)
    find_program(CHESS_LAB_NODE_EXECUTABLE NAMES node nodejs REQUIRED)
    set_target_properties(m0_smoke PROPERTIES SUFFIX ".js")
    target_link_options(m0_smoke PRIVATE "-sENVIRONMENT=node" "-sEXIT_RUNTIME=1")
    add_test(NAME m0_smoke COMMAND "${CHESS_LAB_NODE_EXECUTABLE}" "$<TARGET_FILE:m0_smoke>")
  else()
    add_test(NAME m0_smoke COMMAND m0_smoke)
  endif()
  set_tests_properties(m0_smoke PROPERTIES TIMEOUT 30)
endif()
